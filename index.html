function renderEmployeeView(emp) { const cards = emp.clients.map(client => `

Cliente ${client.slot}

${client.clientName || 'Sin asignar'}

`).join(''); employeeContent.innerHTML = cards; employeeContent.querySelectorAll('button[data-slot]').forEach(btn => { btn.addEventListener('click', () => validateAndMark(emp, btn.dataset.action, Number(btn.dataset.slot))); }); } async function validateAndMark(emp, kind, slot) { if (!navigator.geolocation) { alert('Tu navegador no soporta geolocalización.'); return; } navigator.geolocation.getCurrentPosition(({ coords }) => { const { latitude: lat, longitude: lng } = coords; const client = emp.clients.find(c => c.slot === Number(slot)); if (!client) { alert('Cliente no encontrado.'); return; } const clientLat = Number(client.clientLat), clientLng = Number(client.clientLng); if (!Number.isFinite(clientLat) || !Number.isFinite(clientLng)) { alert('El administrador aún no configuró coordenadas de este cliente.'); return; } const distance = haversine(lat, lng, clientLat, clientLng); client.lastDistance = distance; client.geoValidated = distance <= FIXED_RADIUS; if (!client.geoValidated) { alert(`Estás a ${Math.round(distance)} m del cliente. Debes estar dentro de ${FIXED_RADIUS} m.`); return; } const now = new Date().toLocaleString('en-US'); if (kind === 'checkIn') client.checkIn = now; else client.checkOut = now; addHistory(`${kind === 'checkIn' ? 'Entrada' : 'Salida'} registrada`, `${emp.name} · Cliente ${client.slot}: ${client.clientName || 'Sin nombre'} · ${Math.round(distance)} m.`); renderEmployeeView(emp); }, () => alert('No se pudo obtener la ubicación. Activa GPS y permisos.')); }
